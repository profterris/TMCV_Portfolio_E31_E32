/** * Publication des portfolios E31 / E32 – bac pro MCV * * Ce script reçoit le portfolio envoyé par le bouton « Publier mon portfolio en ligne », * l'enregistre dans ton Google Drive (dossier privé) et le rend consultable par un lien. * Chaque publication est ajoutée automatiquement dans une feuille « Liens des portfolios ». * * Installation (une seule fois) : * 1. Exécuter la fonction « installer » (menu en haut) et accepter les autorisations. * 2. Déployer > Nouveau déploiement > type « Application Web » * Exécuter en tant que : Moi – Qui a accès : Tout le monde * 3. Copier l'adresse de l'application Web (elle se termine par /exec). * * Date limite : dans la feuille « Liens des portfolios E31-E32 », onglet « Réglages », * saisir la date en B2. Après cette date (jour entier compris), plus aucune publication * ni modification n'est acceptée. Case vide : pas de limite. Aucun redéploiement nécessaire. */ const NOM_DOSSIER = 'Portfolios E31-E32 – publiés'; const NOM_FEUILLE = 'Liens des portfolios E31-E32'; const TAILLE_MAX = 45 * 1024 * 1024; // 45 Mo /** À exécuter une fois : crée le dossier et la feuille, et déclenche les autorisations. */ function installer() { const dossier = dossier_(); const feuille = feuille_(); reglages_(); Logger.log('Dossier : ' + dossier.getUrl()); Logger.log('Feuille des liens : ' + feuille.getParent().getUrl()); } /** Réception d'un portfolio publié par un élève. */ function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (!d.html || !d.nom || !d.prenom) return json_({ ok: false, erreur: 'Portfolio incomplet (nom, prénom ou contenu manquant).' }); if (d.html.length > TAILLE_MAX) return json_({ ok: false, erreur: 'Portfolio trop lourd (45 Mo maximum). Remplace les vidéos jointes par des liens.' }); const limite = dateLimite_(); if (limite && new Date() > limite) { return json_({ ok: false, ferme: true, limite: limite.toISOString(), erreur: 'La date limite du ' + texteDate_(limite) + ' est passée : ton portfolio ne peut plus être publié ni modifié. Adresse-toi à ton professeur.' }); } const verrou = LockService.getScriptLock(); verrou.waitLock(30000); try { const props = PropertiesService.getScriptProperties(); let id = String(d.id || ''); let rec = id ? JSON.parse(props.getProperty('p_' + id) || 'null') : null; if (rec && rec.token !== d.token) { rec = null; } // jeton invalide : nouvelle publication if (!rec) { id = Utilities.getUuid().replace(/-/g, '').slice(0, 12); rec = { token: Utilities.getUuid(), cree: new Date().toISOString() }; } const nomFichier = nettoie_(d.nom).toUpperCase() + '_' + nettoie_(d.prenom) + '_portfolio_E31_E32.html'; const fichier = dossier_().createFile(Utilities.newBlob(d.html, 'text/html', nomFichier)); if (rec.fileId) { try { DriveApp.getFileById(rec.fileId).setTrashed(true); } catch (err) {} } rec.fileId = fichier.getId(); rec.titre = 'Portfolio E31 / E32 – ' + d.prenom + ' ' + String(d.nom).toUpperCase(); props.setProperty('p_' + id, JSON.stringify(rec)); const url = ScriptApp.getService().getUrl() + '?p=' + id; noteFeuille_(id, d, url); return json_({ ok: true, id: id, token: rec.token, url: url, limite: limite ? limite.toISOString() : '' }); } finally { verrou.releaseLock(); } } catch (err) { return json_({ ok: false, erreur: String(err && err.message || err) }); } } /** Affichage d'un portfolio publié : …/exec?p=identifiant */ function doGet(e) { const id = e && e.parameter && e.parameter.p; const rec = id ? JSON.parse(PropertiesService.getScriptProperties().getProperty('p_' + id) || 'null') : null; if (!rec) { return HtmlService.createHtmlOutput('

Portfolio introuvable. Vérifie le lien.

') .setTitle('Portfolio introuvable'); } const html = DriveApp.getFileById(rec.fileId).getBlob().getDataAsString('UTF-8'); return HtmlService.createHtmlOutput(html) .setTitle(rec.titre || 'Portfolio E31 / E32') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /* ---------- outils ---------- */ function dossier_() { const props = PropertiesService.getScriptProperties(); const id = props.getProperty('dossier'); if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} } const dossier = DriveApp.createFolder(NOM_DOSSIER); props.setProperty('dossier', dossier.getId()); return dossier; } function feuille_() { const props = PropertiesService.getScriptProperties(); let ss = null; const id = props.getProperty('feuille'); if (id) { try { ss = SpreadsheetApp.openById(id); } catch (e) {} } if (!ss) { ss = SpreadsheetApp.create(NOM_FEUILLE); DriveApp.getFileById(ss.getId()).moveTo(dossier_()); const sh = ss.getSheets()[0]; sh.setName('Liens'); sh.appendRow(['Nom', 'Prénom', 'Classe', 'N° candidat', 'Lien du portfolio', 'Première publication', 'Dernière mise à jour', 'Identifiant']); sh.setFrozenRows(1); sh.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#1f4e79').setFontColor('#ffffff'); sh.setColumnWidth(5, 420); sh.getRange('F:G').setNumberFormat('dd/mm/yyyy hh:mm'); ss.setSpreadsheetTimeZone('Europe/Paris'); props.setProperty('feuille', ss.getId()); } return ss.getSheets()[0]; } /** Onglet « Réglages » : B2 = date limite de publication et de modification. */ function reglages_() { const ss = feuille_().getParent(); let sh = ss.getSheetByName('Réglages'); if (!sh) { sh = ss.insertSheet('Réglages'); sh.getRange('A1:B1').setValues([['Réglage', 'Valeur']]).setFontWeight('bold').setBackground('#1f4e79').setFontColor('#ffffff'); sh.getRange('A2').setValue('Date limite de publication et de modification (incluse)'); sh.getRange('B2').setNumberFormat('dd/mm/yyyy hh:mm').setBackground('#fff2cc'); sh.getRange('A3').setValue('Case vide : pas de limite. Une date seule (sans heure) autorise toute la journée.'); sh.getRange('A3').setFontStyle('italic').setFontColor('#5f6b7a'); sh.setColumnWidth(1, 420); sh.setColumnWidth(2, 160); } return sh; } /** Date limite lue dans « Réglages »!B2 ; null si vide. */ function dateLimite_() { const v = reglages_().getRange('B2').getValue(); if (!(v instanceof Date) || isNaN(v)) return null; const d = new Date(v.getTime()); if (d.getHours() === 0 && d.getMinutes() === 0 && d.getSeconds() === 0) d.setHours(23, 59, 59, 999); return d; } function texteDate_(d) { return Utilities.formatDate(d, 'Europe/Paris', "dd/MM/yyyy 'à' HH:mm"); } function noteFeuille_(id, d, url) { const sh = feuille_(); const lignes = sh.getDataRange().getValues(); const maintenant = new Date(); const num = d.num ? "'" + d.num : ''; for (let r = 1; r < lignes.length; r++) { if (lignes[r][7] === id) { sh.getRange(r + 1, 1, 1, 7).setValues([[String(d.nom).toUpperCase(), d.prenom, d.classe || '', num, url, lignes[r][5], maintenant]]); return; } } sh.appendRow([String(d.nom).toUpperCase(), d.prenom, d.classe || '', num, url, maintenant, maintenant, id]); } function nettoie_(t) { return String(t || '').normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[^A-Za-z0-9-]/g, ''); } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
